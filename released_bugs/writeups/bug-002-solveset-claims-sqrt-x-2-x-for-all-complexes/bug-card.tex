\section*{Bug 2: solveset claims \texorpdfstring{$\sqrt{x^2}=x$}{sqrt(x**2) = x} for all complexes}

\begin{counterexamplebox}
\textbf{Task.} Solve the principal-square-root equation over the complex numbers:
\[
    \sqrt{x^2}=x.
\]

\medskip
\textbf{SymPy's answer.}
\[
    \operatorname{solveset}(\sqrt{x^2}=x,\ x,\ \mathbb{C})=\mathbb{C}.
\]

\medskip
\textbf{Correct answer.}
\[
    \{z:\operatorname{Re}(z)>0\}\cup\{it:t\in\mathbb{R},\ t\ge 0\},
\]
in particular not \(\mathbb{C}\).

\medskip
\textbf{Explanation.} The set contains \(x=-1\), yet \(\sqrt{(-1)^2}=1\ne -1\). Squaring treats \(\sqrt{x^2}=x\) as branch-insensitive, dropping the principal-root branch.

\medskip
\textbf{Likely root cause.} In \nolinkurl{sympy/solvers/solveset.py}, \texttt{\_solve\_radical} accepts the non-finite result produced after \texttt{unrad} turns the equation into \(0=0\), without validating it against the original equation. This is a new instance of a known failure mode.

\medskip
\textbf{Artifact (GitHub).} \bugbundle{bug-002-solveset-claims-sqrt-x-2-x-for-all-complexes}
\end{counterexamplebox}
