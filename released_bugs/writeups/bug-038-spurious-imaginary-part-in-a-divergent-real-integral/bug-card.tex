\section*{Bug 38: Spurious Imaginary Part in a Divergent Real Integral}

\begin{counterexamplebox}
\textbf{Task.} Compute the negative-side improper integral of \(1/|x|\).
\[
  \int_{-1}^{0} \frac{1}{|x|}\,dx
\]

\medskip
\textbf{SymPy's answer.}
\[
  \operatorname{integrate}\!\left(\frac{1}{\operatorname{Abs}(x)},(x,-1,0)\right)
  = \infty + i\pi.
\]

\medskip
\textbf{Correct answer.}
\[
  \int_{-1}^{0} \frac{1}{|x|}\,dx = \infty.
\]

\medskip
\textbf{Explanation.} On \([-1,0)\), \(1/|x|=-1/x\), so
\(\int_{-1}^{-\varepsilon}1/|x|\,dx=-\log(\varepsilon)\to\infty\) as
\(\varepsilon\to0^+\).  The integrand is real and nonnegative, so the finite
\(i\pi\) term is a spurious complex-log branch contribution.

\medskip
\textbf{Likely root cause.} The diagnosis identifies
\texttt{\_eval\_interval} endpoint evaluation for a \texttt{Piecewise}
antiderivative: the negative branch is represented as \(-\log(x)\), and
substitution at \(x=-1\) keeps the principal-log value \(-i\pi\) instead of a
real branch value such as \(-\log(-x)\).  This is a specific new instance of
the already-reported defect in open issue
\href{https://github.com/sympy/sympy/issues/23337}{\SymPy{}~\#23337}
(``Spurious I*pi in improper definite integral'').

\medskip
\textbf{Artifact (GitHub).} \bugbundle{bug-038-spurious-imaginary-part-in-a-divergent-real-integral}
\end{counterexamplebox}
