\section*{Bug 12: inverse Fourier transform drops \texorpdfstring{$|t|$}{Abs(t)}}

\begin{counterexamplebox}
\textbf{Task.} Compute the inverse Fourier transform of the Lorentzian
\[
  F(w)=\frac{2}{1+4\pi^2w^2}.
\]

\medskip
\textbf{SymPy's answer.}
\[
  \operatorname{inverse\_fourier\_transform}(F(w),w,t)=e^{-t}.
\]

\medskip
\textbf{Correct answer.}
\[
  \operatorname{inverse\_fourier\_transform}(F(w),w,t)=e^{-|t|}.
\]

\medskip
\textbf{Explanation.} The input \(F(w)\) is real and even in \(w\), so its
inverse Fourier transform should be real and even in \(t\).  SymPy returns
\(e^{-t}\), which is not even: for example, at \(t=-1\) it gives \(e\), while
the expected value is \(e^{-1}\).  The missing absolute value changes the
answer on the negative half-line.

\medskip
\textbf{Likely root cause.} In
\nolinkurl{sympy/integrals/transforms.py}, \texttt{\_fourier\_transform}
appears to select the first branch of a conditional \texttt{Piecewise} integral,
then the public inverse transform drops the condition under the default
\texttt{noconds=True}.  This is a new instance of a known failure mode. 

\medskip
\textbf{Artifact (GitHub).} \bugbundle{bug-012-inverse-fourier-transform-drops-abs-t}
\end{counterexamplebox}
