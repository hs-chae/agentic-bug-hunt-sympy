\section*{Bug 31: False Real Solutions for \texttt{acos(cos(u)) = -1}}

\begin{counterexamplebox}
\textbf{Task.} Solve the real equation
\[
  \operatorname{acos}(\cos u) = -1.
\]

\medskip
\textbf{SymPy's answer.} \(\operatorname{solveset}(\operatorname{acos}(\cos u)=-1,u,\mathbb{R})\)
returns
\[
  \{2\pi n+1:n\in\mathbb{Z}\}\cup
  \{2\pi n-1+2\pi:n\in\mathbb{Z}\}.
\]

\medskip
\textbf{Correct answer.} The solution set is \(\varnothing\).

\medskip
\textbf{Explanation.} For real \(u\), \(\cos u\in[-1,1]\) and the principal
arccosine has range \([0,\pi]\), so \(\operatorname{acos}(\cos u)\) is never negative.
For example, a returned representative \(u=1\) gives
\(\operatorname{acos}(\cos 1)-(-1)=2\), not zero.

\medskip
\textbf{Likely root cause.} The diagnosis identifies
\nolinkurl{sympy/solvers/solveset.py}: \texttt{\_invert\_real} applies
\(\operatorname{acos}^{-1}=\cos\) to the target before checking that \(-1\) lies in
\(\operatorname{acos}\)'s principal range.  This is a new instance of a known failure mode.

\medskip
\textbf{Artifact (GitHub).} \bugbundle{bug-031-false-real-solutions-for-acos-cos-u-1}
\end{counterexamplebox}
