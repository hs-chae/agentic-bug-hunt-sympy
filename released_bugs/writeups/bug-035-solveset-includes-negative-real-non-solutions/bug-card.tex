\section*{Bug 35: solveset includes negative real non-solutions}

\begin{counterexamplebox}
\textbf{Task.} Solve over the complex numbers:
\[
  \sqrt{x}\sqrt{1/x}=1.
\]

\medskip
\textbf{SymPy's answer.}
\[
  \operatorname{solveset}\!\left(\sqrt{x}\sqrt{1/x}=1,\ x,\ \mathbb{C}\right)
  = \mathbb{C}\setminus\{0\}.
\]

\medskip
\textbf{Correct answer.}
\[
  \mathbb{C}\setminus\{x\in\mathbb{R}:x\le 0\}.
\]

\medskip
\textbf{Explanation.} The returned set contains \(x=-1\), but
\[
  \sqrt{-1}\sqrt{1/(-1)} = i\cdot i = -1 \ne 1.
\]
The principal complex square root is not multiplicative across the negative real
branch cut, so the negative real axis must be excluded.

\medskip
\textbf{Likely root cause.} In \nolinkurl{sympy/solvers/solveset.py},
\texttt{\_solve\_radical} accepts an infinite superset produced by
\texttt{unrad} without rechecking it against the original radical equation. This
is a new instance of a known failure mode.

\medskip
\textbf{Artifact (GitHub).} \bugbundle{bug-035-solveset-includes-negative-real-non-solutions}
\end{counterexamplebox}
