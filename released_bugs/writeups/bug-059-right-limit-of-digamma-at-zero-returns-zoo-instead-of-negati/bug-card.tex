\section*{Bug 59: Right Limit of Digamma at Zero Returns Zoo Instead of Negative Infinity}

\begin{counterexamplebox}
\textbf{Task.} Evaluate the right-hand limit of the order-zero polygamma function at the origin.
\[
\lim_{x \to 0^+} \operatorname{polygamma}(0,x)
\]

\medskip
\textbf{SymPy's answer.}
\[
\operatorname{limit}(\operatorname{polygamma}(0,x),x,0,\operatorname{dir}=+)=\mathtt{zoo}.
\]

\medskip
\textbf{Correct answer.}
\[
\lim_{x \to 0^+} \operatorname{polygamma}(0,x)=-\infty.
\]

\medskip
\textbf{Explanation.} Since \(\operatorname{polygamma}(0,x)=\psi(x)\), the Laurent expansion at zero is \(\psi(x)=-1/x-\gamma+O(x)\). Along \(x \to 0^+\), the leading term tends to \(-\infty\), so \(\mathtt{zoo}\) incorrectly loses the real directional sign.

\medskip
\textbf{Likely root cause.} The diagnosis locates the omission in \texttt{polygamma.\_eval\_as\_leading\_term()}, which leaves \(\operatorname{polygamma}(0,x)\) unchanged instead of exposing the finite-pole leading term \(-1/x\); the limit fallback then evaluates \(\operatorname{polygamma}(0,0)\) as complex infinity. Deduplication finds this duplicates an existing unmerged public fix, SymPy PR~\#29424 (\url{https://github.com/sympy/sympy/pull/29424}), which targets the same symptom and root cause.

\end{counterexamplebox}
