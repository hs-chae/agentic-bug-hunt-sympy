\section*{Bug 60: LerchPhi Left Limit Returns Divergent Endpoint}

\begin{counterexamplebox}
\textbf{Task.} Compute the left-hand limit of the Lerch transcendent:
\[
\lim_{x\to 1^-}\operatorname{lerchphi}(x,1,1).
\]

\medskip
\textbf{SymPy's answer.} \(\SymPy\) returns the unevaluated endpoint
\[
\operatorname{limit}(\operatorname{lerchphi}(x,1,1),x,1,\mathrm{dir}=-)
=\operatorname{lerchphi}(1,1,1).
\]

\medskip
\textbf{Correct answer.} The mathematically correct result is
\[
+\infty.
\]

\medskip
\textbf{Explanation.} For \(0<x<1\),
\[
\operatorname{lerchphi}(x,1,1)=\sum_{n=0}^{\infty}\frac{x^n}{n+1}
=\frac{-\log(1-x)}{x},
\]
so the left-hand limit diverges to \(+\infty\). The returned endpoint represents the harmonic series at \(x=1\), not a finite limit value.

\medskip
\textbf{Likely root cause.} The diagnosis traces the limit through Gruntz, where \texttt{mrv\_leadterm} treats \(\operatorname{lerchphi}(1-1/z,1,1)\) as having coefficient \(\operatorname{lerchphi}(1,1,1)\) and exponent \(0\), instead of exposing the logarithmic divergence available through expansion. This is a new instance of a known failure mode.

\medskip
\textbf{Artifact (GitHub).} \bugbundle{bug-060-lerchphi-left-limit-returns-divergent-endpoint}
\end{counterexamplebox}
