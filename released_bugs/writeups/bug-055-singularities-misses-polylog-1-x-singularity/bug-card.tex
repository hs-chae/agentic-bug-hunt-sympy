\section*{Bug 55: \texorpdfstring{singularities Misses \(\operatorname{polylog}(1,x)\) Singularity}{singularities Misses polylog(1,x) Singularity}}

\begin{counterexamplebox}
\textbf{Task.} Find the complex singularities of the order-one polylogarithm.
\[
\operatorname{singularities}(\operatorname{polylog}(1,x),x,\mathbb{C})
\]

\medskip
\textbf{SymPy's answer.}
\[
\operatorname{singularities}(\operatorname{polylog}(1,x),x,\mathbb{C})=\varnothing .
\]

\medskip
\textbf{Correct answer.}
\[
1 \in \operatorname{singularities}(\operatorname{polylog}(1,x),x,\mathbb{C}) .
\]

\medskip
\textbf{Explanation.} The identity \(\operatorname{polylog}(1,z)=-\log(1-z)\) gives \(\operatorname{polylog}(1,x)=-\log(1-x)\), which has a logarithmic singularity at \(x=1\).  SymPy itself evaluates both \(\operatorname{polylog}(1,1)\) and \(-\log(1-1)\) to \(\mathtt{zoo}\), so the empty set misses a concrete singular point.

\medskip
\textbf{Likely root cause.} The implementation in \texttt{singularities} scans for a hard-coded set of visible singular forms such as powers and logarithms, but it does not handle \(\operatorname{polylog}\) or expand \(\operatorname{polylog}(1,z)\) to \(-\log(1-z)\) before scanning.  This is a new failure mode with no public precedent.

\medskip
\textbf{Artifact (GitHub).} \bugbundle{bug-055-singularities-misses-polylog-1-x-singularity}
\end{counterexamplebox}
