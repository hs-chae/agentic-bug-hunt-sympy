\section*{Bug 48: Core algebraic identities erase incomplete Piecewise undefined branches}

\begin{counterexamplebox}
\textbf{Task.} Evaluate core identities built from an incomplete \texttt{Piecewise}
on its implicit undefined branch.
\[
p=\operatorname{Piecewise}((1,x>0)),\qquad x=-1
\]

\medskip
\textbf{SymPy's answer.}
\[
p(-1)=\mathrm{NaN},\qquad (p\cdot0)(-1)=0,\qquad (p-p)(-1)=0,\qquad
\operatorname{Eq}(p,p)(-1)=\mathrm{True}.
\]

\medskip
\textbf{Correct answer.}
\[
(p\cdot0)(-1)=\mathrm{NaN},\qquad (p-p)(-1)=\mathrm{NaN},\qquad
\operatorname{Eq}(p,p)(-1)\ne\mathrm{True}.
\]

\medskip
\textbf{Explanation.} The expression \(p\) has no branch for \(x\leq0\), and
\SymPy{} already evaluates \(p(-1)\) as \(\mathrm{NaN}\).  The finite-value
identities \(z\cdot0=0\), \(z-z=0\), and \(z=z\) are not valid at a point where
\(z\) is undefined, so these simplifications incorrectly extend the domain.

\medskip
\textbf{Likely root cause.} The diagnosis points to generic core simplification:
\nolinkurl{sympy/core/mul.py} collapses multiplication and cancellation without
checking for incomplete \texttt{Piecewise} operands, while
\nolinkurl{sympy/core/relational.py} turns structurally identical operands into
\texttt{True}.  Deduplication frames this as a new instance of a known failure
mode.

\medskip
\textbf{Artifact (GitHub).} \bugbundle{bug-048-core-algebraic-identities-erase-incomplete-piecewise-undefin}
\end{counterexamplebox}
